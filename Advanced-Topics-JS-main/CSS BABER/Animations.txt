Basic Scenario:
1-See Properties values

*{
margin: 0;
padding: 0;
box-sizing: border-box;
}

.container{
  background-color: aqua;
  border:3px solid black;
  height:100vh;
  width: 100vw;
 
}
.box{
   width: 350px;
  height: 80px; background-color: blue;border: 3px solid yellow;
  border-radius: 10px;
  margin: 10px;
  position: relative;
  animation-name: rigidMovementt;  //The name of the animation (from your @keyframes)
  animation-duration: 5s;  //How long one cycle lasts
  animation-iteration-count: infinite;       How many times to repeat
  animation-delay: 1s;                        How long to wait before starting
  animation-timing-function: linear;          How the speed changes during animation
  animation-direction: alternate;            //ek cycle complete krny k bad isy wapis kis trah ana chahiye
}

@keyframes rigidMovement {
  from{
    top: 0;
    left: 0;
  }
  to{
    top: 10px;
    left: 900px;
  }
}


2-animation-fill-mode
//jab ek cycle complete hu jai tab usy kaha rukna chahiye,forwards or backward

.box{
  width: 350px;height: 80px; background-color: blue;border: 3px solid yellow;
  border-radius: 10px;
  margin: 10px;
  position: relative;
  animation-name: rigidMovementt;
  animation-duration: 5s;
  /* animation-iteration-count: infinite;  */
  animation-delay: 1s;
  animation-timing-function: linear;
  /* animation-direction: alternate; */
  animation-fill-mode: forwards;
}
