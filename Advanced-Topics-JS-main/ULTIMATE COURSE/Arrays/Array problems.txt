=>Find k numbers
let arrayValue = [12,35,1,10,34,1];
function FindKNumbers(arr,k){
let size=arr.length;
if (k>size){
  k=k%size;
}
let rotated=arr.splice(size-k,size);
arr.unshift(...rotated);
return arr;
}

console.log(FindKNumbers(arrayValue,4));

Note:
Modulus is the Correct Approach in case of k greater
Mathematical Correctness: Rotating an array by k positions where k > size is equivalent to rotating by k % size positions

Example: Rotating [1,2,3] by 4 positions = rotating by 1 position (since 4 % 3 = 1)

find the smallest difference of two values in all array
let data = [23, 77, 21, 44, 100, 99];

function findClosestPair(arr) {
  // Step 1: Sort the array
  arr.sort((a, b) => a - b);

  let minDiff = Infinity; //Infinity in JavaScript is just a special number that is larger than any other number.
  let pair = [];

  // Step 2: Compare neighbors
  for (let i = 0; i < arr.length - 1; i++) {
    let diff = arr[i + 1] - arr[i];
    if (diff < minDiff) {
      minDiff = diff;
      pair = [arr[i], arr[i + 1]];
    }
  }

  return { pair, difference: minDiff };
}

console.log(findClosestPair(data));

Instead of Infinity, you could start with the difference of the first two elements:

function largestPairSum(arr) {
  // Sort the array
  arr.sort((a, b) => b - a); // descending order

  // The two biggest numbers will be at index 0 and 1
  let max1 = arr[0];
  let max2 = arr[1];

  return { pair: [max1, max2], sum: max1 + max2 };
}

function data(array, pair) {
  let arraychunk = [];
  for (let i = 0; i < array.length; i += pair) {
    arraychunk.push(array.slice(i, i + pair));

  return arraychunk;
}

let names=["Tayyab Jamshaid","Hafiz Amir Attari Khan"]


let data=names.map((item)=>{
  //trim Removes extra spaces from start and end. e.g "  Tayyab Jamshaid  ".trim() ===>"Tayyab Jamshaid"
  //split(/\s+/)
//  /\s+/ is a regular expression
//\s → matches any whitespace(space,tab,newLine)
//+ → means one or more
//So it splits the string no matter how many spaces are between words.
//e.g "Muhammad   Tayyab  Bin   Jamshaid".split(/\s+/)    Result:["Muhammad", "Tayyab", "Bin", "Jamshaid"]
let val=item.trim().split(/\s+/);
let firstName=val[0];
let lastName=val.length>1?val[val.length-1]:undefined;
//parts.slice(1, -1)
//1 → start from index 1 (skip first name)
//-1 → stop before the last element
let middleName=val.length>2?val.slice(1,-1).join(""):undefined

return {firstName,lastName,middleName}
})
